T2：高精度二进制小数转十进制

题目描述

:::success
给定一个合法的二进制小数（格式为 “整数部分。小数部分”），需将其转换为十进制数并输出。由于二进制小数的小数部分长度可能超过 200 位（远超<font style="color:rgba(0, 0, 0, 0.85) !important;">long double</font>的精度范围，直接使用浮点数会导致严重精度丢失），因此必须通过字符串逐位解析结合高精度计算实现转换，确保结果准确性。

:::

输入说明
  1. 输入为单个字符串，仅包含字符<font style="color:rgb(0, 0, 0);">'0'</font>、<font style="color:rgb(0, 0, 0);">'1'</font>和<font style="color:rgb(0, 0, 0);">'.'</font>，代表合法的二进制小数；
  2. 字符串仅含1 个小数点，且小数点前后均至少有 1 位字符（排除 “0.”“1.”“.0”“.1” 等无效格式）；
  3. 整数部分仅由<font style="color:rgb(0, 0, 0);">'0'</font>或<font style="color:rgb(0, 0, 0);">'1'</font>组成（如 “0”“1”“101”“1110” 等），小数部分仅由<font style="color:rgb(0, 0, 0);">'0'</font>或<font style="color:rgb(0, 0, 0);">'1'</font>组成；
  4. 小数部分长度范围：<font style="color:rgb(0, 0, 0);">1 ≤ 小数部分长度 ≤ 200</font>（确保超过常规浮点数精度，强制要求高精度处理）；
  5. 无需处理无效输入（如含非 0/1 字符、多个小数点等）。
输出说明
  1. 有限小数处理：若转换结果为有限小数（二进制小数有限的充要条件是：小数部分无无限循环的 1，即分母仅含 2 的因子），需精确输出所有有效数字，末尾无多余 0，小数点后无数字时不输出小数点（例：二进制 “1.000” 输出 “1”，而非 “1.0”）；
  2. 无限小数处理：若转换结果为无限小数（如二进制 “0.010101...” 对应十进制 1/3），则输出保留小数点后 15 位，第 16 位数字四舍五入（若第 16 位≥5，则第 15 位加 1，依次进位，如 0.3333333333333334→0.333333333333333）；
  3. 输出结果禁止使用科学计数法，仅以 “整数” 或 “整数。小数” 形式呈现。
示例解析

示例 1：基础有限小数

  • 输入：<font style="color:rgb(0, 0, 0);">0.1101</font>
  • 输出：<font style="color:rgb(0, 0, 0);">0.8125</font>

示例 2：含非 0 整数部分

  • 输入：<font style="color:rgb(0, 0, 0);">10.101</font>
  • 输出：<font style="color:rgb(0, 0, 0);">2.625</font>

示例 3：长小数部分（10 位）

  • 输入：<font style="color:rgb(0, 0, 0);">0.1101110110</font>
  • 输出：<font style="color:rgb(0, 0, 0);">0.865234375</font>
题解：

该题目核心考察高精度计算能力，需突破常规浮点数和整数的限制，具体考点包括：

  1. 字符串拆分：将输入按小数点拆分为整数部分字符串和小数部分字符串，逐位提取二进制位；
  2. 高精度整数计算（二进制转十进制）：
    • 对于长整数部分（如 100 位 1），需用数组 / 字符串存储十进制 digits（如<font style="color:rgb(0, 0, 0);">digits[0]</font>存个位，<font style="color:rgb(0, 0, 0);">digits[1]</font>存十位），逐位计算 “当前值 ×2 + 二进制位”，避免溢出；
  3. 高精度小数计算（累加 1/2^k）：
    • 用数组存储小数点后的十进制 digits（如<font style="color:rgb(0, 0, 0);">dec_frac[0]</font>存小数点后第 1 位，<font style="color:rgb(0, 0, 0);">dec_frac[1]</font>存第 2 位）；
    • 对每一位二进制小数<font style="color:rgb(0, 0, 0);">b_k</font>（小数点后第 k 位），计算<font style="color:rgb(0, 0, 0);">b_k × (1/2^k)</font>，通过 “初始值 ÷2” 的循环实现（如 1/2^3 = (1÷2)÷2÷2），并将结果累加到小数 digits 数组中；
  4. 精度控制逻辑：
    • 判断有限 / 无限小数：二进制小数有限 ↔ 小数部分无无限循环的 1（或输入小数部分已遍历完且无后续位，需根据数学性质判断）；
    • 去尾随 0：从小数 digits 末尾向前删除连续的 0，若删除后无小数位，则不输出小数点；
    • 四舍五入：对需保留的第 15 位小数，检查第 16 位，若≥5 则进位（需处理连续进位，如 0.9999999999999995→1.000000000000000）；
  5. 边界情况处理：
    • 整数部分为 0（如 “0.1”）；
    • 整数部分为极大值（如 “111...111”，100 个 1）；
    • 小数部分长度达 200 位（需高效处理循环计算，避免超时）。
#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

// 高精度加法：将两个大整数相加
vector<int> addBigNumbers(const vector<int>& a, const vector<int>& b) {
    vector<int> result;
    int carry = 0;
    int i = 0;
    
    while (i < a.size() || i < b.size() || carry > 0) {
        int sum = carry;
        if (i < a.size()) sum += a[i];
        if (i < b.size()) sum += b[i];
        
        carry = sum / 10;
        result.push_back(sum % 10);
        i++;
    }
    
    return result;
}

// 高精度乘法：大整数乘以2
vector<int> multiplyByTwo(const vector<int>& num) {
    vector<int> result;
    int carry = 0;
    
    for (int digit : num) {
        int product = digit * 2 + carry;
        result.push_back(product % 10);
        carry = product / 10;
    }
    
    if (carry > 0) {
        result.push_back(carry);
    }
    
    return result;
}

// 高精度除法：小数部分除以2
vector<int> divideFractionByTwo(const vector<int>& frac, int precision) {
    vector<int> result;
    int carry = 0;
    
    for (int i = 0; i < precision; i++) {
        int current = carry * 10;
        if (i < frac.size()) {
            current += frac[i];
        }
        
        result.push_back(current / 2);
        carry = (current % 2);
    }
    
    return result;
}

// 高精度加法：小数部分相加
vector<int> addFractions(const vector<int>& a, const vector<int>& b, int precision) {
    vector<int> result;
    int carry = 0;
    
    for (int i = 0; i < precision; i++) {
        int sum = carry;
        if (i < a.size()) sum += a[i];
        if (i < b.size()) sum += b[i];
        
        result.push_back(sum % 10);
        carry = sum / 10;
    }
    
    return result;
}

// 处理四舍五入
void roundFraction(vector<int>& frac, int precision) {
    if (frac.size() <= precision) return;
    
    int roundDigit = frac[precision];
    if (roundDigit >= 5) {
        int carry = 1;
        for (int i = precision - 1; i >= 0 && carry > 0; i--) {
            int sum = frac[i] + carry;
            frac[i] = sum % 10;
            carry = sum / 10;
        }
    }
    
    if (frac.size() > precision) {
        frac.resize(precision);
    }
}

// 移除小数部分末尾的零
void removeTrailingZeros(vector<int>& frac) {
    while (!frac.empty() && frac.back() == 0) {
        frac.pop_back();
    }
}

int main() {
    string binary;
    cin >> binary;
    
    // 分割整数部分和小数部分
    size_t dotPos = binary.find('.');
    string intPartStr = binary.substr(0, dotPos);
    string fracPartStr = binary.substr(dotPos + 1);
    
    const int MAX_PRECISION = 16; // 用于四舍五入的精度
    
    // 处理整数部分
    vector<int> intPart;
    for (char c : intPartStr) {
        // 整数部分计算: current = current * 2 + (c - '0')
        vector<int> temp = multiplyByTwo(intPart);
        if (c == '1') {
            intPart = addBigNumbers(temp, {1});
        } else {
            intPart = temp;
        }
    }
    
    // 处理小数部分
    vector<int> fracPart(MAX_PRECISION, 0);
    vector<int> currentFrac(MAX_PRECISION, 0);
    
    for (int i = 0; i < fracPartStr.size(); i++) {
        // 当前位的值是 (bit) / 2^(i+1)
        // 计算 1/2^(i+1)
        if (i == 0) {
            currentFrac[0] = 5; // 1/2 = 0.5
            for (int j = 1; j < MAX_PRECISION; j++) {
                currentFrac[j] = 0;
            }
        } else {
            currentFrac = divideFractionByTwo(currentFrac, MAX_PRECISION);
        }
        
        // 如果当前位是1，加到结果中
        if (fracPartStr[i] == '1') {
            fracPart = addFractions(fracPart, currentFrac, MAX_PRECISION);
        }
    }
    
    // 处理四舍五入到15位小数
    roundFraction(fracPart, 15);
    
    // 移除末尾的零
    removeTrailingZeros(fracPart);
    
    // 输出结果
    // 输出整数部分
    if (intPart.empty()) {
        cout << "0";
    } else {
        for (auto it = intPart.rbegin(); it != intPart.rend(); ++it) {
            cout << *it;
        }
    }
    
    // 输出小数部分（如果有的话）
    if (!fracPart.empty()) {
        cout << ".";
        for (int digit : fracPart) {
            cout << digit;
        }
    }
    
    cout << endl;
    
    return 0;
}

T3：模板矩阵类的加法运算符重载实现

题目描述

:::success
定义一个模板矩阵类 **<font style="color:rgb(0, 0, 0);">Matrix<T, ROW, COL></font>**（其中 <font style="color:rgba(0, 0, 0, 0.85) !important;">T</font> 为数据类型，<font style="color:rgba(0, 0, 0, 0.85) !important;">ROW</font> 为矩阵行数，<font style="color:rgba(0, 0, 0, 0.85) !important;">COL</font> 为矩阵列数，<font style="color:rgba(0, 0, 0, 0.85) !important;">ROW</font> 和 <font style="color:rgba(0, 0, 0, 0.85) !important;">COL</font> 为编译期固定的正整数），用于存储 <font style="color:rgba(0, 0, 0, 0.85) !important;">ROW</font> 行、<font style="color:rgba(0, 0, 0, 0.85) !important;">COL</font> 列的同类型数据。

需为该模板类重载加法运算符 **<font style="color:rgb(0, 0, 0);">+</font>**，使不同数据类型、不同固定维度的矩阵支持合法的加法运算，具体规则如下：

  1. 运算前提：两个相加的矩阵必须为同类型、同维度（即 <font style="color:rgb(0, 0, 0);">T</font> 相同、<font style="color:rgb(0, 0, 0);">ROW</font> 相同、<font style="color:rgb(0, 0, 0);">COL</font> 相同），否则视为非法运算，需提示 “维度或类型不匹配，无法相加”；
  2. 不同类型的加法规则：
    • 当 <font style="color:rgb(0, 0, 0);">T = int</font> 且 <font style="color:rgb(0, 0, 0);">ROW=2、COL=3</font> 时（即 <font style="color:rgb(0, 0, 0);">Matrix<int, 2, 3></font>）：加法为对应位置元素的整数相加（如矩阵 A 的 (i,j) 元素 + 矩阵 B 的 (i,j) 元素，结果作为新矩阵的 (i,j) 元素）；
    • 当 <font style="color:rgb(0, 0, 0);">T = double</font> 且 <font style="color:rgb(0, 0, 0);">ROW=3、COL=2</font> 时（即 <font style="color:rgb(0, 0, 0);">Matrix<double, 3, 2></font>）：加法为对应位置元素的浮点数相加，结果保留 6 位小数（若末尾为 0 可省略，如 2.500000 输出 2.5）；
    • 当 <font style="color:rgb(0, 0, 0);">T = string</font> 且 <font style="color:rgb(0, 0, 0);">ROW=3、COL=3</font> 时（即 <font style="color:rgb(0, 0, 0);">Matrix<String, 3, 3></font>）：加法为对应位置字符串的拼接（如矩阵 A 的 (i,j) 字符串 + 矩阵 B 的 (i,j) 字符串，结果作为新矩阵的 (i,j) 元素）；
  3. 需实现矩阵的输入函数（从标准输入读取矩阵元素）和输出函数（将矩阵元素按行输出，每行元素用空格分隔，行末无多余空格），支持上述三种特定类型矩阵的输入输出。

:::

#include <iostream>
#include <vector>
#include <string>
#include <iomanip>
#include <sstream>

using namespace std;

// 模板矩阵类定义
template <typename T, int ROW, int COL>
class Matrix {
private:
    vector<vector<T>> data;

public:
    // 构造函数
    Matrix() : data(ROW, vector<T>(COL, T())) {}

    // 输入函数
    void input() {
        for (int i = 0; i < ROW; ++i) {
            for (int j = 0; j < COL; ++j) {
                cin >> data[i][j];
            }
        }
    }

    // 输出函数
    void output() const {
        for (int i = 0; i < ROW; ++i) {
            for (int j = 0; j < COL; ++j) {
                // 针对double类型特殊处理输出格式
                if constexpr (is_same_v<T, double>) {
                    double val = data[i][j];
                    // 检查是否为整数
                    if (val == floor(val)) {
                        cout << (long long)val;
                    } else {
                        // 先按6位小数输出到字符串
                        stringstream ss;
                        ss << fixed << setprecision(6) << val;
                        string s = ss.str();
                        // 移除末尾多余的0
                        s.erase(s.find_last_not_of('0') + 1, string::npos);
                        // 如果最后一个字符是小数点，也移除
                        if (s.back() == '.') {
                            s.pop_back();
                        }
                        cout << s;
                    }
                } else {
                    // 对于int和string类型直接输出
                    cout << data[i][j];
                }
                // 控制空格，行末无多余空格
                if (j != COL - 1) {
                    cout << " ";
                }
            }
            cout << endl;
        }
    }

    // 加法运算符重载
    Matrix<T, ROW, COL> operator+(const Matrix<T, ROW, COL>& other) const {
        Matrix<T, ROW, COL> result;
        for (int i = 0; i < ROW; ++i) {
            for (int j = 0; j < COL; ++j) {
                result.data[i][j] = this->data[i][j] + other.data[i][j];
            }
        }
        return result;
    }
};

int main() {
    // 测试用例1：int类型矩阵 (2x3)
    Matrix<int, 2, 3> intMat1, intMat2;
    intMat1.input();
    intMat2.input();
    Matrix<int, 2, 3> intResult = intMat1 + intMat2;
    intResult.output();
    cout << endl;  // 测试用例间空一行

    // 测试用例2：double类型矩阵 (3x2)
    Matrix<double, 3, 2> doubleMat1, doubleMat2;
    doubleMat1.input();
    doubleMat2.input();
    Matrix<double, 3, 2> doubleResult = doubleMat1 + doubleMat2;
    doubleResult.output();
    cout << endl;  // 测试用例间空一行

    // 测试用例3：string类型矩阵 (3x3)
    Matrix<string, 3, 3> stringMat1, stringMat2;
    stringMat1.input();
    stringMat2.input();
    Matrix<string, 3, 3> stringResult = stringMat1 + stringMat2;
    stringResult.output();

    return 0;
}
